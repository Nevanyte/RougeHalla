var drawerProof = new Dictionary<string, object>();
var drawerChecks = new List<object>();
string drawerFailure = null;
int drawerCaptureFrame = -1;

bool DrawerCheck(string name, bool passed, object facts)
{
    drawerChecks.Add(new { name, passed, facts });
    if (!passed && drawerFailure == null) drawerFailure = name;
    return passed;
}
void WriteDrawerProof(string status)
{
    drawerProof["status"] = status;
    drawerProof["failure"] = drawerFailure;
    drawerProof["checks"] = drawerChecks;
    drawerProof["sourceSha"] = sourceSha;
    drawerProof["screen"] = new { width = Screen.width, height = Screen.height, display = Display.main != null ? Display.main.renderingWidth : 0,
        frame = Time.frameCount, time = Time.realtimeSinceStartupAsDouble };
    Directory.CreateDirectory(outputFolder);
    File.WriteAllText(Path.Combine(outputFolder, "drawer-proof.json"), JsonConvert.SerializeObject(drawerProof, Formatting.Indented));
}
RaycastResult[] DrawerHits(Vector2 position)
{
    var es = EventSystem.current;
    if (es == null) return new RaycastResult[0];
    var data = new PointerEventData(es) { position = position };
    var hits = new List<RaycastResult>();
    es.RaycastAll(data, hits);
    return hits.ToArray();
}
object DrawerHitFacts(RaycastResult[] hits)
{
    return hits.Select(h => new { path = h.gameObject != null ? PathOf(h.gameObject.transform) : null,
        module = h.module != null ? h.module.GetType().FullName : null, depth = h.depth, sortingLayer = h.sortingLayer, sortingOrder = h.sortingOrder }).ToArray();
}
Button DrawerButton(GameObject hit)
{
    return hit != null ? hit.GetComponentInParent<Button>() : null;
}
object DrawerMaterialInfo(Material material)
{
    if (material == null) return new { present = false };
    var shader = material.shader;
    var passes = new List<object>();
    int count = material.passCount;
    for (int i = 0; i < count; i++)
    {
        bool compiled = false;
        string error = null;
        try { compiled = UnityEditor.ShaderUtil.IsPassCompiled(material, i); }
        catch (Exception ex) { error = ex.GetType().Name + ": " + ex.Message; }
        passes.Add(new { index = i, name = material.GetPassName(i), compiled, error });
    }
    return new { present = true, name = material.name, shader = shader != null ? shader.name : null,
        shaderSupported = shader != null && shader.isSupported, passCount = count, passes };
}
object DrawerGraphicInfo(Graphic graphic)
{
    if (graphic == null) return new { present = false };
    var renderer = graphic.canvasRenderer;
    var main = new List<object>();
    var pop = new List<object>();
    if (renderer != null)
    {
        for (int i = 0; i < renderer.materialCount; i++) main.Add(DrawerMaterialInfo(renderer.GetMaterial(i)));
        if (renderer.hasPopInstruction)
            for (int i = 0; i < renderer.materialCount; i++) pop.Add(DrawerMaterialInfo(renderer.GetPopMaterial(i)));
    }
    return new { present = true, path = PathOf(graphic.transform), type = graphic.GetType().Name,
        active = graphic.isActiveAndEnabled, activeInHierarchy = graphic.gameObject.activeInHierarchy,
        bounds = graphic.transform is RectTransform ? BoundsOf((RectTransform)graphic.transform) : null,
        depth = graphic.depth, cull = renderer != null && renderer.cull, hasRenderer = renderer != null,
        materialForRendering = DrawerMaterialInfo(graphic.materialForRendering), rendererMaterials = main,
        hasPopInstruction = renderer != null && renderer.hasPopInstruction, popMaterials = pop,
        raycastTarget = graphic.raycastTarget };
}
bool DrawerMaterialReady(Material material)
{
    if (material == null || material.shader == null || !material.shader.isSupported || material.passCount <= 0) return false;
    for (int i = 0; i < material.passCount; i++)
        if (!UnityEditor.ShaderUtil.IsPassCompiled(material, i)) return false;
    return true;
}
bool DrawerGraphicReady(Graphic graphic)
{
    if (graphic == null || !graphic.isActiveAndEnabled || graphic.depth < 0 || graphic.canvasRenderer == null || graphic.canvasRenderer.cull) return false;
    var renderer = graphic.canvasRenderer;
    if (!DrawerMaterialReady(graphic.materialForRendering)) return false;
    for (int i = 0; i < renderer.materialCount; i++) if (!DrawerMaterialReady(renderer.GetMaterial(i))) return false;
    if (renderer.hasPopInstruction)
        for (int i = 0; i < renderer.materialCount; i++) if (!DrawerMaterialReady(renderer.GetPopMaterial(i))) return false;
    return true;
}
object DrawerRenderContext(BuildDrawer drawer, GameObject card)
{
    var canvas = drawer.GetComponentInChildren<Canvas>(true);
    var root = canvas != null ? canvas.rootCanvas : null;
    var raycaster = root != null ? root.GetComponent<GraphicRaycaster>() : null;
    var view = drawer.Scroll != null ? drawer.Scroll.viewport : null;
    var viewportImage = view != null ? view.GetComponent<Image>() : null;
    var bg = card != null ? card.GetComponent<Image>() : null;
    var glyph = card != null && card.transform.Find("CardGlyph") != null ? card.transform.Find("CardGlyph").GetComponent<Image>() : null;
    var drawerRoot = drawer.transform.Find("MatchUI/BuildDrawer");
    var title = drawerRoot != null ? drawerRoot.Find("DrawerTitle") : null;
    var count = drawerRoot != null ? drawerRoot.Find("CardCount") : null;
    var close = drawerRoot != null ? drawerRoot.Find("Close") : null;
    return new { screen = new { width = Screen.width, height = Screen.height },
        canvas = root != null ? new { name = root.name, mode = root.renderMode.ToString(), targetDisplay = root.targetDisplay,
            pixelRect = new[] { root.pixelRect.x, root.pixelRect.y, root.pixelRect.width, root.pixelRect.height },
            scaleFactor = root.scaleFactor, eventCamera = root.worldCamera != null ? root.worldCamera.name : null } : null,
        raycasterEventCamera = raycaster != null && raycaster.eventCamera != null ? raycaster.eventCamera.name : null,
        viewport = view != null ? BoundsOf(view) : null,
        viewportHasMask = view != null && view.GetComponent<Mask>() != null,
        viewportShowMaskGraphic = view != null && view.GetComponent<Mask>() != null && view.GetComponent<Mask>().showMaskGraphic,
        viewportMask = DrawerGraphicInfo(viewportImage), cardBackground = DrawerGraphicInfo(bg), cardGlyph = DrawerGraphicInfo(glyph),
        drawerTitle = DrawerGraphicInfo(title != null ? title.GetComponent<Graphic>() : null),
        cardCount = DrawerGraphicInfo(count != null ? count.GetComponent<Graphic>() : null),
        closeControl = DrawerGraphicInfo(close != null ? close.GetComponent<Graphic>() : null),
        cardGraphics = card != null ? card.GetComponentsInChildren<Graphic>(true).Select(DrawerGraphicInfo).ToArray() : new object[0] };
}
bool DrawerPointInside(float[] bounds, Vector2 p)
{
    return bounds != null && p.x >= bounds[0] && p.x <= bounds[2] && p.y >= bounds[1] && p.y <= bounds[3];
}
bool FindVisibleHit(BuildDrawer drawer, out Button button, out GameObject hitObject, out Vector2 point, out RaycastResult[] hits)
{
    button = null; hitObject = null; point = Vector2.zero; hits = new RaycastResult[0];
    if (drawer == null || drawer.Scroll == null || drawer.Scroll.viewport == null) return false;
    float[] v = BoundsOf(drawer.Scroll.viewport);
    foreach (Transform child in drawer.Content)
    {
        var card = child as RectTransform;
        if (card == null || !card.gameObject.activeInHierarchy) continue;
        float[] b = BoundsOf(card);
        point = new Vector2((b[0] + b[2]) * .5f, (b[1] + b[3]) * .5f);
        if (!DrawerPointInside(v, point)) continue;
        hits = DrawerHits(point);
        if (hits.Length == 0 || hits[0].gameObject == null) continue;
        var resolved = DrawerButton(hits[0].gameObject);
        if (resolved == null || resolved.gameObject != card.gameObject || !resolved.IsInteractable()) continue;
        button = resolved; hitObject = hits[0].gameObject; return true;
    }
    return false;
}
bool FindDifferentVisibleHit(BuildDrawer drawer, Button differentFrom, out Button button, out GameObject hitObject, out Vector2 point, out RaycastResult[] hits)
{
    button = null; hitObject = null; point = Vector2.zero; hits = new RaycastResult[0];
    if (drawer == null || drawer.Scroll == null || drawer.Scroll.viewport == null) return false;
    float[] v = BoundsOf(drawer.Scroll.viewport);
    foreach (Transform child in drawer.Content)
    {
        var card = child as RectTransform;
        if (card == null || !card.gameObject.activeInHierarchy) continue;
        var candidate = card.GetComponent<Button>();
        if (candidate == null || candidate == differentFrom || !candidate.IsInteractable()) continue;
        float[] b = BoundsOf(card);
        point = new Vector2((b[0] + b[2]) * .5f, (b[1] + b[3]) * .5f);
        if (!DrawerPointInside(v, point)) continue;
        hits = DrawerHits(point);
        if (hits.Length == 0 || hits[0].gameObject == null || DrawerButton(hits[0].gameObject) != candidate) continue;
        button = candidate; hitObject = hits[0].gameObject; return true;
    }
    return false;
}
bool FindClippedOutsidePoint(BuildDrawer drawer, out Button button, out Vector2 point, out float[] cardBounds, out RaycastResult[] hits)
{
    button = null; point = Vector2.zero; cardBounds = null; hits = new RaycastResult[0];
    if (drawer == null || drawer.Scroll == null || drawer.Scroll.viewport == null) return false;
    float[] v = BoundsOf(drawer.Scroll.viewport);
    foreach (Transform child in drawer.Content)
    {
        var rect = child as RectTransform;
        if (rect == null || !rect.gameObject.activeInHierarchy) continue;
        var b = BoundsOf(rect);
        float ix0 = Mathf.Max(b[0], v[0]), ix1 = Mathf.Min(b[2], v[2]);
        float iy0 = Mathf.Max(b[1], v[1]), iy1 = Mathf.Min(b[3], v[3]);
        if (ix1 - ix0 <= 1f || iy1 - iy0 <= 1f) continue;
        var candidateButton = rect.GetComponent<Button>();
        if (candidateButton == null) continue;
        if (b[1] < v[1] - .5f) point = new Vector2((ix0 + ix1) * .5f, (b[1] + Mathf.Min(b[3], v[1])) * .5f);
        else if (b[3] > v[3] + .5f) point = new Vector2((ix0 + ix1) * .5f, (Mathf.Max(b[1], v[3]) + b[3]) * .5f);
        else if (b[0] < v[0] - .5f) point = new Vector2((b[0] + Mathf.Min(b[2], v[0])) * .5f, (iy0 + iy1) * .5f);
        else if (b[2] > v[2] + .5f) point = new Vector2((Mathf.Max(b[0], v[2]) + b[2]) * .5f, (iy0 + iy1) * .5f);
        else continue;
        if (!DrawerPointInside(b, point) || DrawerPointInside(v, point)) continue;
        button = candidateButton; cardBounds = b; hits = DrawerHits(point); return true;
    }
    return false;
}
Dictionary<string, object> DrawerClipProbe(BuildDrawer drawer)
{
    Button card; Vector2 point; float[] bounds; RaycastResult[] hits;
    if (!FindClippedOutsidePoint(drawer, out card, out point, out bounds, out hits))
        return new Dictionary<string, object> { ["found"] = false, ["reason"] = "No partially clipped card with an in-card point outside the viewport was found." };
    bool rejected = !hits.Any(h => h.gameObject != null && DrawerButton(h.gameObject) == card);
    return new Dictionary<string, object> { ["found"] = true, ["card"] = PathOf(card.transform), ["point"] = new[] { point.x, point.y },
        ["cardBounds"] = bounds, ["viewportBounds"] = BoundsOf(drawer.Scroll.viewport), ["rejected"] = rejected, ["hits"] = DrawerHitFacts(hits) };
}
void OpenDrawerByRenderedHit()
{
    try
    {
        var drawer = review.Hud.GetComponent<BuildDrawer>();
        var es = EventSystem.current;
        var detailsTransform = review.Hud.transform.Find("MatchUI/PlayerHUD/BuildStrip/BuildDetails");
        var button = detailsTransform != null ? detailsTransform.GetComponent<Button>() : null;
        if (drawer == null || es == null || button == null) throw new InvalidOperationException("BuildDrawer, EventSystem, or BuildDetails button missing.");
        if (drawer.IsVisible) throw new InvalidOperationException("Expected drawer closed before initial open.");
        var rect = button.transform as RectTransform;
        var bounds = BoundsOf(rect);
        var point = new Vector2((bounds[0] + bounds[2]) * .5f, (bounds[1] + bounds[3]) * .5f);
        var hits = DrawerHits(point);
        var top = hits.Length > 0 ? hits[0].gameObject : null;
        var resolved = DrawerButton(top);
        bool topIsButton = resolved == button;
        bool depthReady = button.GetComponent<Graphic>() != null && button.GetComponent<Graphic>().depth >= 0;
        bool rayHit = hits.Length > 0 && top != null && topIsButton && depthReady;
        if (!DrawerCheck("BuildDetails.center-resolves-rendered-button", rayHit,
            new { point = new[] { point.x, point.y }, bounds, topIsButton, resolved = resolved != null ? PathOf(resolved.transform) : null,
                depth = button.GetComponent<Graphic>() != null ? button.GetComponent<Graphic>().depth : -1, hits = DrawerHitFacts(hits) }))
            throw new InvalidOperationException("BuildDetails center did not resolve to its processed button hit.");
        var data = new PointerEventData(es) { position = point, pressPosition = point, button = PointerEventData.InputButton.Left, clickCount = 1 };
        data.pointerPressRaycast = hits[0]; data.pointerCurrentRaycast = hits[0];
        var executed = ExecuteEvents.ExecuteHierarchy<IPointerClickHandler>(top, data, ExecuteEvents.pointerClickHandler);
        bool opened = drawer.IsVisible;
        var help = review.Hud.transform.Find("MatchUI/Help");
        var settings = review.Hud.transform.Find("MatchUI/Settings");
        var leave = review.Hud.transform.Find("MatchUI/Leave");
        bool headersHidden = help != null && settings != null && leave != null && !help.gameObject.activeInHierarchy &&
            !settings.gameObject.activeInHierarchy && !leave.gameObject.activeInHierarchy;
        DrawerCheck("BuildDetails.routed-click-opens-and-hides-global-header", executed == button.gameObject && opened && headersHidden,
            new { executed = executed != null ? PathOf(executed.transform) : null, opened, headersHidden });
        if (!opened || !headersHidden) throw new InvalidOperationException("Rendered hit-resolved BuildDetails click did not open drawer/hide header.");
        drawerProof["open"] = new { point = new[] { point.x, point.y }, bounds, resolvedTop = PathOf(top.transform),
            executed = PathOf(executed.transform), render = DrawerRenderContext(drawer, drawer.Content.Find("Card00").gameObject) };
    }
    catch (Exception ex)
    {
        if (drawerFailure == null) drawerFailure = "open-exception: " + ex.GetType().Name + ": " + ex.Message;
        WriteDrawerProof("failed-before-capture");
        Debug.LogError("[drawer-proof] " + drawerFailure);
        throw;
    }
}

        var buildHits = DrawerHits(buildPoint); var buildTop = buildHits.Length > 0 ? buildHits[0].gameObject : null;
        bool buildResolved = buildTop != null && DrawerButton(buildTop) == buildButton && buildButton.GetComponent<Graphic>().depth >= 0;
        if (!DrawerCheck("reopen-center-hit-resolves", buildResolved, new { point = new[] { buildPoint.x, buildPoint.y }, hits = DrawerHitFacts(buildHits) })) yield break;
        var buildData = new PointerEventData(EventSystem.current) { position = buildPoint, pressPosition = buildPoint,
            button = PointerEventData.InputButton.Left, clickCount = 1, pointerCurrentRaycast = buildHits[0], pointerPressRaycast = buildHits[0] };
        var buildHandler = ExecuteEvents.ExecuteHierarchy<IPointerClickHandler>(buildTop, buildData, ExecuteEvents.pointerClickHandler);
        yield return null; yield return new WaitForEndOfFrame(); Canvas.ForceUpdateCanvases();
        bool persisted = drawer.IsVisible && drawer.SelectedIndex == selectedBeforeClose && Mathf.Abs(drawer.Scroll.verticalNormalizedPosition - scrollBeforeClose) <= .0001f;
        if (!DrawerCheck("routed-reopen-preserves-scroll-selection", buildHandler == buildButton.gameObject && persisted,
            new { handler = buildHandler != null ? PathOf(buildHandler.transform) : null, drawerVisible = drawer.IsVisible,
                selectedBeforeClose, selectedAfterReopen = drawer.SelectedIndex, scrollBeforeClose, scrollAfterReopen = drawer.Scroll.verticalNormalizedPosition })) yield break;
        drawerProof["renderFinal"] = DrawerRenderContext(drawer, clicked.gameObject);
        DrawerCheck("final-frame-advancing", Time.frameCount > drawerCaptureFrame,
            new { frame = Time.frameCount, captureFrame = drawerCaptureFrame });
        complete = drawerFailure == null;
    }
    finally
    {
        if (!complete && drawerFailure == null) drawerFailure = "Verification stopped before all checks completed.";
        WriteDrawerProof(complete ? "passed" : "failed-or-incomplete");
        if (!complete) Debug.LogError("[drawer-proof] " + drawerFailure);
    }
}
float MaxBoundsDelta(float[] a, float[] b)
{
    if (a == null || b == null || a.Length != 4 || b.Length != 4) return float.PositiveInfinity;
    float delta = 0f; for (int i = 0; i < 4; i++) delta = Mathf.Max(delta, Mathf.Abs(a[i] - b[i])); return delta;
}
